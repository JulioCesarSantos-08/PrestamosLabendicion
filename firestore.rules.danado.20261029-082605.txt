
$ErrorActionPreference = "Stop"

$ruta = "firestore.rules"
$reglas = Get-Content $ruta -Raw -Encoding UTF8

if ($reglas.Contains("function cobroEmpleadoPermitido(")) {
    throw "Esta actualizacion ya esta instalada."
}

if (!$reglas.Contains("match /prestamoControl/{prestamoId}")) {
    throw "Falta instalar prestamoControl en las reglas."
}

$funcion = @'
    function cobroEmpleadoPermitido(prestamoId) {
      return empleadoPuedeCobrar(prestamoId)
        && exists(
          /databases/$(database)/documents/prestamoControl/$(prestamoId)
        )
        && get(
          /databases/$(database)/documents/prestamoControl/$(prestamoId)
        ).data.estado == "activo"
        && get(
          /databases/$(database)/documents/prestamoControl/$(prestamoId)
        ).data.empleadoId == request.auth.uid;
    }

'@

$marcador = "    match /saldosPrestamo/{prestamoId} {"

if (!$reglas.Contains($marcador)) {
    throw "No se encontro la seccion de saldos."
}

$reglas = $reglas.Replace($marcador, $funcion + $marcador)

$reemplazos = @(
    @{
        Antes = 'allow create: if activeAdmin()'
        Despues = 'allow create: if (activeAdmin() || cobroEmpleadoPermitido(prestamoId))'
        Seccion = 'saldosPrestamo'
    },
    @{
        Antes = 'allow update: if activeAdmin()'
        Despues = 'allow update: if (activeAdmin() || cobroEmpleadoPermitido(prestamoId))'
        Seccion = 'saldosPrestamo'
    },
    @{
        Antes = 'allow create: if activeAdmin()'
        Despues = 'allow create: if (activeAdmin() || cobroEmpleadoPermitido(request.resource.data.prestamoId))'
        Seccion = 'pagos'
    },
    @{
        Antes = 'allow create: if activeAdmin()'
        Despues = 'allow create: if (activeAdmin() || cobroEmpleadoPermitido(request.resource.data.prestamoId))'
        Seccion = 'recibosAbono'
    }
)

$secciones = @(
    @{
        Nombre = 'saldosPrestamo'
        Inicio = '    match /saldosPrestamo/{prestamoId} {'
        Fin = '    match /pagos/{pagoId} {'
    },
    @{
        Nombre = 'pagos'
        Inicio = '    match /pagos/{pagoId} {'
        Fin = '    match /recibosAbono/{reciboId} {'
    },
    @{
        Nombre = 'recibosAbono'
        Inicio = '    match /recibosAbono/{reciboId} {'
        Fin = '    match /{document=**} {'
    }
)

foreach ($seccion in $secciones) {
    $inicio = $reglas.IndexOf($seccion.Inicio)
    $fin = $reglas.IndexOf($seccion.Fin, $inicio)

    if ($inicio -lt 0 -or $fin -lt 0) {
        throw "No se encontro la seccion $($seccion.Nombre)."
    }

    $bloque = $reglas.Substring($inicio, $fin - $inicio)

    foreach ($item in $reemplazos | Where-Object {
        $_.Seccion -eq $seccion.Nombre
    }) {
        if (!$bloque.Contains($item.Antes)) {
            throw "No coincide una regla de $($seccion.Nombre)."
        }

        $bloque = $bloque.Replace($item.Antes, $item.Despues)
    }

    $reglas = $reglas.Substring(0, $inicio) +
        $bloque +
        $reglas.Substring($fin)
}

$marca = Get-Date -Format "yyyyMMdd-HHmmss"
Copy-Item $ruta "$ruta.$marca.bak"

Set-Content $ruta $reglas -Encoding UTF8

Write-Host ""
Write-Host "REGLAS DE COBRANZA ACTUALIZADAS" -ForegroundColor Green
Write-Host "Respaldo creado: $marca"
Write-Host "Ahora debes publicarlas en Firebase."
